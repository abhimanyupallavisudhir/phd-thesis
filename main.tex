% ============================================================
% Chapter-loading mechanism.
%
% Each chapter lives in chapters/<name>.tex, with an optional
% chapter-local macro file macros/<name>.tex. Chapter macros are
% loaded inside a group, so different chapters may redefine the
% same command (e.g. \bools, \claudehaiku) without clashing.
% The \clearpage before \endgroup flushes floats while the
% chapter-local macros are still in scope.
% ============================================================
\NewDocumentCommand{\thesischapter}{m}{%
  \begingroup
  \InputIfFileExists{macros/#1.tex}{}{}%
  \input{chapters/#1.tex}%
  \clearpage
  \endgroup
}

% The ToC/LoF/LoT are typeset in the front matter, before any chapter's
% macros are in scope — but section titles and captions written to
% main.toc/.lof/.lot may use chapter-local macros. So we typeset them
% inside a group that pre-loads ALL chapter macro files with tolerant
% (last-wins) definitions; the few cross-chapter conflicts (e.g. \bools)
% don't matter for headings/captions.
\NewDocumentCommand{\withallchaptermacros}{+m}{%
  \begingroup
  \let\NewDocumentCommand\DeclareDocumentCommand
  \let\RenewDocumentCommand\DeclareDocumentCommand
  \let\newcommand\DeclareRobustCommand
  \let\newcommandx\providecommandx
  \let\algnewcommand\DeclareRobustCommand
  \InputIfFileExists{macros/market-rl.tex}{}{}%
  \InputIfFileExists{macros/agency.tex}{}{}%
  \InputIfFileExists{macros/infonomy.tex}{}{}%
  \InputIfFileExists{macros/solib.tex}{}{}%
  \InputIfFileExists{macros/nonvf.tex}{}{}%
  \InputIfFileExists{macros/consistency.tex}{}{}%
  #1%
  \endgroup
}
